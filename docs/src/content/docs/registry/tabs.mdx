## Usage

```tsx
<Tabs value={tab} onValueChange={setTab} defaultValue="one">
  <TabsList>
    <TabsTrigger value="one">One</TabsTrigger>
    <TabsTrigger value="two">Two</TabsTrigger>
  </TabsList>
  <TabsContent value="one">First tab</TabsContent>
```
